 71def rk4_solve(rhs, initial, eta_array, step_size):
 72    '''Fixed-step RK4 for y' = rhs(eta, y) from initial over eta_array.
 73    Returns the state at every point of eta_array, shape (len(eta_array), len(initial))'''
 74    BLASIUS_RK4 = np.zeros((len(eta_array), len(initial)))
 75    BLASIUS_RK4[0] = np.array(initial)
 76
 77    # RK4 loop
 78    for i in range(len(eta_array)-1):
 79        eta = eta_array[i]
 80        k1 = rhs(eta, BLASIUS_RK4[i])
 81        BLASIUS_trial_2 = BLASIUS_RK4[i] + 0.5*step_size*k1
 82        k2 = rhs(eta+0.5*step_size, BLASIUS_trial_2)
 83        BLASIUS_trial_3 = BLASIUS_RK4[i] + 0.5*step_size*k2
 84        k3 = rhs(eta+0.5*step_size, BLASIUS_trial_3)
 85        BLASIUS_trial_4 = BLASIUS_RK4[i] + step_size*k3
 86        k4 = rhs(eta+step_size, BLASIUS_trial_4)
 87        WEIGHTED_RATE = (k1+2*k2+2*k3+k4)/6
 88
 89
 90        blasius_rk4_next = BLASIUS_RK4[i]+step_size*WEIGHTED_RATE
 91        BLASIUS_RK4[i+1] = blasius_rk4_next
 92    return BLASIUS_RK4

104def secant_for_shooting(f, s_old, s_init, max_iter=MAX_ITERATIONS, precision=TOLERANCE):
105    """
106    Finds a root of the function f(x) using the secant method.
107
108    Args:
109        f: Shooting function to be reiterated
110        s_init: Initial guess for the root
111        s_old: Previous initial guess
112        precision: Stopping criteria determined by the user
113
114    Returns:
115        s_new: Root of the function f(x)
116        num_iter: Number of iterations taken to find the root
117    """
118    f_old = f(s_old)
119    if not np.isfinite(f_old):
120        raise ValueError("Secant method failed to converge")
121    f_init = f(s_init)
122    if not np.isfinite(f_init):
123        raise ValueError("Secant method failed to converge")
124
125    f_array = np.array([f_old, f_init])
126    s_array = np.array([s_old, s_init])
127
128    for k in range(max_iter):
129        f_old, f_init = f_array[k], f_array[k+1]
130        s_old, s_init = s_array[k], s_array[k+1]
131        if np.abs(f_init-f_old) < np.finfo(float).eps:
132            raise ValueError("Secant method failed to converge")
133
134        s_new = s_init - f_init*(s_init - s_old) / (f_init - f_old)  # estimate first guess of s_new
135
136        if not np.isfinite(s_new) or s_new <= 0:
137            raise ValueError("Secant method failed to converge")
138
139        f_new = f(s_new)
140        s_array = np.append(s_array, s_new)
141        f_array = np.append(f_array, f_new)
142
143        if not np.isfinite(f_new):
144            raise ValueError("Secant method failed to converge")
145
146        if np.abs(f_new) < precision:
147            return s_new, k+1, s_array, f_array
148
149    # return s_new as the root and iteration counter
150    return s_new, k+1, s_array, f_array
