src.swd_model.identifiability_screen

Identifiability screen for the planned inverse PINN.

Three questions, asked before any Blasius, von Karman, RK4 or PINN work is built:

  1. If only the friction part of the drag is shifted, does the board's acceleration change by enough, at the ~15 Hz sample times of the paper's instrumented board (mmc1), to be seen above that IMU's noise?
  2. Does that change look different from the change caused by shifting the starting speed?
  3. How precisely could a least-squares fit recover the friction multiplier k from noisy data sampled like this?

The forward model is trajectory (forward Euler, the default_shortboard hydroscan polar, constant slope, raised-cosine roll) with one change: a multiplier k on the friction term of the drag. Nothing here gives a pass/fail verdict - it prints the numbers, and writes them to SWD/data/default_shortboard/computed-data/identifiability_screen/.

Run

python -m src.swd_model.identifiability_screen # stops before a Monte Carlo over 30 min python -m src.swd_model.identifiability_screen --allow-long # runs it regardless

 95@dataclass(frozen=True)
 96class FrictionScaledPolar(tr.BoardPolar):
 97    """A :class:`trajectory.BoardPolar` whose friction drag is multiplied by ``k``.
 98
 99    Attributes
100    ----------
101    k : float
102        Multiplier on the friction part of the drag only. Lift and planing drag stay SWD's.
103    """
104
105    k: float = 1.0
106
107    def drag(self, v: float, roll: np.ndarray | float) -> np.ndarray:
108        """Total drag in newtons, with only the friction term scaled by ``k``.
109
110        Parameters
111        ----------
112        v : float
113            Board speed, m/s.
114        roll : float or numpy.ndarray
115            Roll angle(s), rad.
116
117        Returns
118        -------
119        numpy.ndarray
120            Planing drag plus ``k`` times friction drag, N.
121        """
122        # Planing is defined as drag_n - friction_n, so scaling friction_n itself would
123        # have moved planing by the same amount the other way. Scale the term instead.
124        fric = np.interp(np.abs(roll), self.roll_rad, self.friction_n)
125        plan = np.interp(np.abs(roll), self.roll_rad, self.drag_n) - fric
126        r = v / self.v_ref
127        return plan * r**tr.EXP_PLANING + self.k * fric * r**tr.EXP_FRICTION

130def simulate(
131    polar: tr.BoardPolar,
132    k: float = 1.0,
133    v0: float | None = None,
134    duration_s: float = 3.0,
135    dt: float = 0.01,
136    t_cover: float | None = None,
137    slope_deg: float = SLOPE_DEG,
138    mass_kg: float = 80.0,
139    board_mass_kg: float = 1.741,
140    roll_max_deg: float = 35.0,
141) -> pd.DataFrame:
142    """One carving turn, as :func:`trajectory.simulate`, with ``k``, ``v0`` and grid length free.
143
144    The loop mirrors ``trajectory.simulate`` line for line, so ``k = 1`` with the default
145    ``v0`` reproduces it exactly (``tests/test_identifiability_screen.py`` holds it to that).
146
147    Parameters
148    ----------
149    polar : trajectory.BoardPolar
150        SWD's lift and drag polar.
151    k : float
152        Friction multiplier.
153    v0 : float, optional
154        Starting speed, m/s. Default: the equilibrium speed, as in ``trajectory.simulate``.
155    duration_s : float
156        Length of the carve; it sets the raised-cosine roll period.
157    dt : float
158        Euler step, s.
159    t_cover : float, optional
160        Last time the run is compared at. The grid is then extended past it, because
161        ``np.arange`` stops short of its end. The roll period stays ``duration_s``.
162    slope_deg, mass_kg, board_mass_kg, roll_max_deg : float
163        As in ``trajectory.simulate``; the slope defaults to the author's ``SLOPE_DEG``.
164
165    Returns
166    -------
167    pandas.DataFrame
168        ``timestamp``, ``speed_ms``, ``roll_rad`` and the path-frame ``accel_{x,y,z}_ms2``.
169    """
170    p = FrictionScaledPolar(**{**vars(polar), "k": k})
171    m = mass_kg + board_mass_kg
172    # Two extra steps put the last compared sample between central differences rather
173    # than on np.gradient's one-sided end.
174    end = duration_s if t_cover is None else max(duration_s, t_cover + 2.0 * dt)
175    t = np.arange(0.0, end, dt)
176    if t_cover is not None and t[-1] < t_cover:
177        raise ValueError(f"grid ends at {t[-1]} s, before the last sample at {t_cover} s")
178    phi = np.radians(roll_max_deg) * 0.5 * (1.0 - np.cos(2.0 * np.pi * t / duration_s))
179
180    v = np.empty_like(t)
181    v[0] = tr.equilibrium_speed(p, m, 0.0) if v0 is None else v0
182    drive = m * tr.G * np.sin(np.radians(slope_deg))
183    for i in range(1, t.size):
184        d = float(p.drag(v[i - 1], phi[i - 1]))
185        v[i] = max(v[i - 1] + dt * (drive - d) / m, 0.05)
186
187    r = tr.turn_radius(v, phi)
188    lift = np.array([float(p.lift(vi, pi)) for vi, pi in zip(v, phi)])
189    return pd.DataFrame(
190        {
191            "timestamp": t,
192            "speed_ms": v,
193            "roll_rad": phi,
194            "accel_x_ms2": np.gradient(v, dt),
195            "accel_y_ms2": v**2 / r,
196            "accel_z_ms2": (lift - m * tr.G) / m,
197        }
198    )

201def load_mmc1(path: Path = MMC1) -> pd.DataFrame:
202    """Parse the paper's board sample (mmc1) as ``SOURCE.md`` describes it.
203
204    Parameters
205    ----------
206    path : pathlib.Path
207        The archive as downloaded; it is read in place, never extracted.
208
209    Returns
210    -------
211    pandas.DataFrame
212        148 rows in ascending time: ``t_s`` from 0, ``sg1..sg5``, ``ax_g, ay_g, az_g``
213        (g, gravity included) and the gyroscope ``rx, ry, rz``.
214
215    Raises
216    ------
217    ValueError
218        If any data row does not have the 12 fields.
219    """
220    with zipfile.ZipFile(path) as z:
221        text = z.read("Board_Sample.txt").decode("ascii")
222    # Every other line is a ';'-prefixed hex echo of the line above it - no new information.
223    rows = [ln.split(";") for ln in text.splitlines() if ln.strip() and not ln.startswith(";")]
224    bad = [i for i, r in enumerate(rows) if len(r) != len(MMC1_COLUMNS)]
225    if bad:
226        raise ValueError(f"mmc1 rows {bad[:5]} do not have {len(MMC1_COLUMNS)} fields")
227    df = pd.DataFrame([[float(x) for x in r] for r in rows], columns=MMC1_COLUMNS)
228    # The board counter (ms) runs backwards: reverse it and start the clock at zero.
229    df = df.iloc[::-1].reset_index(drop=True)
230    df.insert(0, "t_s", (df.t_ms - df.t_ms.iloc[0]) / 1000.0)
231    return df.drop(columns="t_ms")

253def at_samples(df: pd.DataFrame, t: np.ndarray) -> np.ndarray:
254    """Linearly interpolate a run's three accelerations onto sample times.
255
256    Parameters
257    ----------
258    df : pandas.DataFrame
259        A run from :func:`simulate`.
260    t : numpy.ndarray
261        Sample times, s, inside the run's grid.
262
263    Returns
264    -------
265    numpy.ndarray
266        Shape ``(len(t), 3)``: accel x, y, z in m/s^2.
267    """
268    ts = df.timestamp.to_numpy()
269    t = np.asarray(t)
270    if t.dtype.kind not in "iuf":
271        raise ValueError("sample times must be numbers (a bool is not)")
272    t = t.astype(float)
273    if t.ndim != 1 or t.size == 0 or not np.all(np.isfinite(t)):
274        raise ValueError("sample times must be a non-empty 1-D array of finite values")
275    # np.interp clamps outside its grid, which would silently hold the last value; min/max, not the ends,
276    # so an unsorted t cannot slip a sample past the check
277    if t.min() < ts[0] or t.max() > ts[-1]:
278        raise ValueError(f"sample times {t.min()}-{t.max()} s fall outside the run {ts[0]}-{ts[-1]} s")
279    return np.column_stack([np.interp(t, ts, df[c].to_numpy()) for c in ACCEL])

305    polar: tr.BoardPolar, duration_s: float, dt: float, t: np.ndarray, v0_true: float
306) -> dict[str, np.ndarray]:

309    Parameters

311    polar : trajectory.BoardPolar

332def choose_dt(
333    polar: tr.BoardPolar, duration_s: float, t: np.ndarray, v0_true: float
334) -> tuple[float, list[dict], dict[str, np.ndarray]]:
335    """Halve dt from ``DT0`` until the delta-a curves stop moving (step 3).
336
337    A step passes when every curve (4 runs x 3 channels) changes by at most 1 % of its
338    own maximum. By Eq 64, ``a_y = v^2/R = g tan(phi)`` does not depend on speed, so the
339    a_y curves are round-off, and "1 % of round-off" would never settle; the absolute
340    floor ``DT_ABS_FLOOR`` covers that case. The finer dt of the passing pair is used.
341
342    Parameters
343    ----------
344    polar : trajectory.BoardPolar
345        SWD's polar.
346    duration_s : float
347        Carve length, s.
348    t : numpy.ndarray
349        Sample times, s.
350    v0_true : float
351        Baseline starting speed, m/s.
352
353    Returns
354    -------
355    dt : float
356        The step used.
357    log : list of dict
358        One entry per halving: the pair, the worst curve's change against its allowance, and
359        the same for the worst x or z curve alone (``*_xz``), so a step decided by a_y
360        round-off is visible as such.
361    d : dict
362        The delta-a curves at ``dt``.
363
364    Raises
365    ------
366    RuntimeError
367        If ``DT_MAX_HALVINGS`` halvings do not converge.
368    """
369    dt, coarse, log = DT0, deltas(polar, duration_s, DT0, t, v0_true), []
370    for _ in range(DT_MAX_HALVINGS):
371        fine = deltas(polar, duration_s, dt / 2, t, v0_true)
372        worst, worst_xz = None, None
373        for name in SHIFTED:
374            for c, ch in enumerate(CHANNELS):
375                change = float(np.max(np.abs(fine[name][:, c] - coarse[name][:, c])))
376                peak = float(np.max(np.abs(fine[name][:, c])))
377                allowed = max(DT_REL_TOL * peak, DT_ABS_FLOOR)
378                e = dict(run=name, channel=ch, change=change, peak=peak, ratio=change / allowed)
379                if worst is None or e["ratio"] > worst["ratio"]:
380                    worst = e
381                if ch != "y" and (worst_xz is None or e["ratio"] > worst_xz["ratio"]):
382                    worst_xz = e
383        log.append(dict(dt_coarse=dt, dt_fine=dt / 2, **worst, **{f"{k}_xz": v for k, v in worst_xz.items()}))
384        if worst["ratio"] <= 1.0:
385            return dt / 2, log, fine
386        dt, coarse = dt / 2, fine
387    raise RuntimeError(f"dt did not converge after {DT_MAX_HALVINGS} halvings: {log[-1]}")

390def quiet_window(mmc1: pd.DataFrame, length: int = QUIET_LENGTH) -> dict:
391    """The quietest stretch of mmc1's accelerometer, as a noise upper bound (step 5).
392
393    Each window of ``length`` consecutive samples has a least-squares line in time removed
394    from every axis; the window with the lowest summed residual variance wins. One window
395    serves all three axes.
396
397    Parameters
398    ----------
399    mmc1 : pandas.DataFrame
400        Output of :func:`load_mmc1`.
401    length : int
402        Window length in samples.
403
404    Returns
405    -------
406    dict
407        ``start`` index, ``t0``/``t1`` in s, per-axis ``sigma`` (m/s^2, ddof = 2 for the two
408        fitted line parameters) and per-axis lag-1 autocorrelation ``lag1`` of the residual.
409    """
410    t = mmc1.t_s.to_numpy()
411    acc = mmc1[["ax_g", "ay_g", "az_g"]].to_numpy() * tr.G  # g, gravity included -> m/s^2
412    best = None
413    for i in range(len(t) - length + 1):
414        tt, aa = t[i : i + length], acc[i : i + length]
415        resid = np.column_stack([aa[:, c] - np.polyval(np.polyfit(tt, aa[:, c], 1), tt) for c in range(3)])
416        score = float((resid**2).sum())
417        if best is None or score < best[0]:
418            best = (score, i, resid)
419    _, i, resid = best
420    return dict(
421        start=i,
422        t0=float(t[i]),
423        t1=float(t[i + length - 1]),
424        sigma=np.sqrt((resid**2).sum(axis=0) / (length - 2)),
425        lag1=(resid[:-1] * resid[1:]).sum(axis=0) / (resid**2).sum(axis=0),
426    )

445def correlation(a: np.ndarray, b: np.ndarray) -> float:

463def standard_error(kind: str, value: float, n: int) -> float:
464    """Standard error of a Monte Carlo statistic over n draws, by the usual normal-theory approximations.
465
466    Printed beside each statistic (the author's choice, AI-USE.md section 4, 2026-10-28), because "settled < 5 %"
467    is a stopping rule, not a statement of precision.
468
469    Parameters
470    ----------
471    kind : {'mean', 'sd', 'fraction', 'corr'}
472        Which statistic: a mean (pass the draws' sd as value), a standard deviation (pass the sd), a fraction (pass
473        p) or a correlation (pass r).
474    value : float
475        As above. NaN gives NaN.
476    n : int
477        Draws behind the statistic: the converged fits for k and V0, every fit for a fraction.
478
479    Returns
480    -------
481    float
482        sd/sqrt(n) for a mean; sd/sqrt(2(n - 1)) for an sd; sqrt(p(1 - p)/n) for a fraction;
483        (1 - r^2)/sqrt(n - 1) for a correlation. NaN when n is too small.
484
485    Raises
486    ------
487    ValueError
488        If kind is not one of the four.
489    """
490    if kind == "mean":
491        return float(value / np.sqrt(n)) if n > 0 else float("nan")
492    if kind == "sd":
493        return float(value / np.sqrt(2 * (n - 1))) if n > 1 else float("nan")
494    if kind == "fraction":
495        return float(np.sqrt(value * (1 - value) / n)) if n > 0 else float("nan")
496    if kind == "corr":
497        return float((1 - value**2) / np.sqrt(n - 1)) if n > 1 else float("nan")
498    raise ValueError(f"kind must be 'mean', 'sd', 'fraction' or 'corr', got {kind!r}")

520        ``(len(t), 3)`` accelerations being fitted, m/s^2.
521    sigma : numpy.ndarray
522        Per-channel noise standard deviation, m/s^2.
523    t : numpy.ndarray
524        Sample times, s.
525    duration_s, dt : float
526        Carve length and Euler step, s.
527    v0_fixed : float or None
528        Starting speed held fixed (set a), or None to fit it (set b).
529
530    Returns
531    -------
532    numpy.ndarray
533        Flattened ``(model - data) / sigma``.
534    """
535    k, v0 = (params[0], v0_fixed) if v0_fixed is not None else (params[0], params[1])
536    model = at_samples(simulate(polar, k=k, v0=v0, duration_s=duration_s, dt=dt, t_cover=t[-1]), t)
537    return ((model - data) / sigma).ravel()

540def fit(
541    polar: tr.BoardPolar,
542    data: np.ndarray,
543    sigma: np.ndarray,
544    t: np.ndarray,
545    duration_s: float,
546    dt: float,
547    v0_true: float,
548    joint: bool,
549) -> tuple[float, float, bool]:
550    """Fit ``k`` (set a) or ``k`` and ``v0`` jointly (set b), from deliberately wrong starts.
551
552    Parameters
553    ----------
554    polar : trajectory.BoardPolar
555        SWD's polar.
556    data : numpy.ndarray
557        ``(len(t), 3)`` accelerations, m/s^2.
558    sigma : numpy.ndarray
559        Per-channel noise standard deviation, m/s^2.
560    t : numpy.ndarray
561        Sample times, s.
562    duration_s, dt : float
563        Carve length and Euler step, s.
564    v0_true : float
565        True starting speed: held fixed in set a, perturbed for the start in set b.
566    joint : bool
567        True for set b.
568
569    Returns
570    -------
571    k_fit : float
572        Fitted friction multiplier.
573    v0_fit : float
574        Fitted starting speed (NaN for set a).
575    converged : bool
576        ``least_squares``'s own success flag.
577    """
578    # Start from the shifted values, not the truth, so a fit cannot pass by never moving.
579    x0 = [PLANTED * (1 + FRICTION_SHIFT)] + ([v0_true * (1 + V0_SHIFT)] if joint else [])
580    res = least_squares(
581        residuals, x0, bounds=(0.0, np.inf),  # TRF keeps iterates strictly inside, so k, v0 > 0
582        args=(polar, data, sigma, t, duration_s, dt, None if joint else v0_true),
583    )
584    return float(res.x[0]), float(res.x[1]) if joint else float("nan"), bool(res.success)

587def monte_carlo(
588    polar: tr.BoardPolar,
589    clean: np.ndarray,
590    noise: np.ndarray,
591    sigma: np.ndarray,
592    t: np.ndarray,
593    duration_s: float,
594    dt: float,
595    v0_true: float,
596    joint: bool,
597) -> tuple[list[tuple[float, float, bool]], dict]:
598    """Fit noisy copies of the baseline, doubling N until the spread of ``k`` settles.
599
600    Draw ``i`` is always ``noise[i]``, so the first N of 2N are the same fits (nested).
601
602    Parameters
603    ----------
604    polar : trajectory.BoardPolar
605        SWD's polar.
606    clean : numpy.ndarray
607        Baseline accelerations at the sample times, ``(len(t), 3)``.
608    noise : numpy.ndarray
609        Standard-normal draws, ``(N_MAX, len(t), 3)``; scaled here by ``sigma``.
610    sigma : numpy.ndarray
611        Per-channel noise standard deviation, m/s^2.
612    t : numpy.ndarray
613        Sample times, s.
614    duration_s, dt : float
615        Carve length and Euler step, s.
616    v0_true : float
617        True starting speed, m/s.
618    joint : bool
619        True for set b.
620
621    Returns
622    -------
623    fits : list of tuple
624        ``(k_fit, v0_fit, converged)`` per draw, in draw order.
625    stop : dict
626        ``n``, ``n2``, ``sd_n``, ``sd_n2`` from the last doubling, and whether it ``settled``.
627    """
628    fits: list[tuple[float, float, bool]] = []
629
630    def sd_k(n: int) -> float:
631        """Standard deviation (ddof 1) of the converged fitted k among the first ``n`` draws."""
632        k = [f[0] for f in fits[:n] if f[2]]
633        return float(np.std(k, ddof=1)) if len(k) > 1 else float("nan")
634
635    n = N0
636    while True:
637        while len(fits) < 2 * n:
638            i = len(fits)
639            fits.append(fit(polar, clean + noise[i] * sigma, sigma, t, duration_s, dt, v0_true, joint))
640        sd_n, sd_n2 = sd_k(n), sd_k(2 * n)
641        # sd_n = 0 (no spread at all) is not "settled"; the live screen guards it the same way
642        settled = sd_n > 0 and abs(sd_n2 - sd_n) / sd_n < STD_REL_TOL
643        if settled or 2 * n >= N_MAX:
644            return fits, dict(n=n, n2=2 * n, sd_n=sd_n, sd_n2=sd_n2, settled=settled)
645        n *= 2
